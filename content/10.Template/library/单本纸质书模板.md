**包含此书的书单**
```dataviewjs
// 获取当前笔记的“书单”字段
const categories = dv.current().书单;

// 检查是否为数组类型
if (!Array.isArray(categories)) {
  dv.paragraph("暂无包含本书的书单");
} else {
  // 过滤出存在的笔记并生成链接
  const validLinks = categories
    .filter(cat => dv.page(`书单/${cat}`))
    .map(cat => dv.fileLink(`书单/${cat}`));

  // 显示结果
  if (validLinks.length > 0) {
    dv.list(validLinks);
  } else {
    dv.paragraph("没有找到对应的书单笔记，该书单索引尚未创建");
  }
}
```
